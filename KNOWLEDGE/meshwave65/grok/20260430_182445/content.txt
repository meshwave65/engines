// INSERT TASK (VERSÃO CORRIGIDA - Direto no Supabase)

async function insertTask() {
  const agentSelect = document.querySelector("#tab2 select")?.value || "meshwave65";
  const agentOverride = document.querySelector("#tab2 input[placeholder='agent override']")?.value?.trim();
  const link = document.querySelector("#tab2 input[placeholder='https://...']")?.value?.trim();

  if (!link) return alert("Link é obrigatório!");

  if (!SESSION.logged || !USER.id) {
    return alert("Você precisa estar logado para inserir tasks.");
  }

  const slug = extractSlugFromUrl(link);
  const llm_provider = extractLLMProvider(link);

  const payload = {
    user_name: USER.user_name,
    agente: agentOverride || agentSelect,
    full_url: link,
    session_user_id: USER.id,
    slug: slug,
    llm_provider: llm_provider,
    // Adicione outros campos obrigatórios/defaults da sua tabela aqui
    // status: "STAGED", created_at: new Date().toISOString(), etc.
  };

  console.log("🔄 Tentando inserir:", payload);

  const { data, error } = await supabase
    .from("appsofia_tasks")
    .insert([payload])
    .select();   // .select() é bom para debug

  if (error) {
    console.error("Erro Supabase insert:", error);
    alert("Erro ao inserir task:\n" + (error.message || error));
    return;
  }

  console.log("✅ Task inserida com sucesso:", data);
  alert("Task inserida com sucesso!");
  
  // Limpa o formulário (opcional)
  document.querySelector("#tab2 input[placeholder='https://...']").value = "";
  if (document.querySelector("#tab2 input[placeholder='agent override']")) {
    document.querySelector("#tab2 input[placeholder='agent override']").value = "";
  }

  loadTasks(); // recarrega a lista
}
