  const map = {
    DONE: "green",
    STAGED: "orange",
    PROCESSING: "blue",
    FAIL: "red"

  return map[status] || "gray";
}

async function loadFiles() {
  if (!USER) return;

  const res = await window.files.list(USER.user_name, CURRENT_PATH);

  if (!res.ok) {
    console.error("FILES LIST ERROR:", res.error);
    return;
  }

  updatePathUI();
  renderFiles(res.data);
}

function renderFiles(items) {
  const el = document.getElementById("files");
  el.innerHTML = "";

  items.forEach(item => {
    const row = document.createElement("div");

    row.style.padding = "6px";
    row.style.cursor = "pointer";
    row.style.borderBottom = "1px solid #eee";

    const isDir = item.isDir ?? item.type === "dir";

    if (isDir) {
      row.innerHTML = 📁 ${item.name};

      row.onclick = () => {
        CURRENT_PATH = CURRENT_PATH
          ? ${CURRENT_PATH}/${item.name}
          : item.name;

        loadFiles();
      };

    } else {
      row.innerHTML = 📄 ${item.name};

      row.onclick = () => openFile(item.path);
    }

    el.appendChild(row);
  });
}

async function openFile(path) {
  const res = await window.files.open(USER.user_name, path);

  if (!res.ok) {
    console.error("OPEN FILE ERROR:", res.error);
    return;
  }

  const viewer = document.getElementById("fileContent");
  viewer.textContent = res.content || "";
}

function goBack() {
  if (!CURRENT_PATH) return;

  const parts = CURRENT_PATH.split("/").filter(Boolean);
  parts.pop();

  CURRENT_PATH = parts.join("/");

  loadFiles();
}

function updatePathUI() {
  const el = document.getElementById("currentPath");
  if (!el) return;

  el.innerText = "/" + (CURRENT_PATH || "");
}

function extractLLMProvider(url = "") {
  try {
    const host = new URL(url).hostname.toLowerCase()
    if (host.includes("manus.im")) return "manus"
    if (host.includes("chatgpt.com")) return "chatgpt"
    if (host.includes("grok.com")) return "grok"
  } catch {}

  const u = url.toLowerCase()
  if (u.includes("manus")) return "manus"
  if (u.includes("chatgpt")) return "chatgpt"
  if (u.includes("grok")) return "grok"

  return null
}

app.post("/auth/login", async (req, res) => {
  const { email, password } = req.body

  try {
    let { data: client } = await supabase
      .from("clients")
      .select("email, user_name, full_name")
      .eq("user_name", email)
      .single()

    if (!client) {
      const r = await supabase
        .from("clients")
        .select("email, user_name, full_name")
        .eq("email", email)
        .single()

      client = r.data
    }

    if (!client) return res.json({ ok: false, error: "user not found" })

    const { data, error } =
      await supabase.auth.signInWithPassword({
        email: client.email,
        password
      })

    if (error) return res.json({ ok: false, error: error.message })

    res.json({
      ok: true,
      user: {
        id: data.user.id,
        email: data.user.email,
        user_name: client.user_name,
        full_name: client.full_name
      }
    })

  } catch (err) {
    res.json({ ok: false, error: err.message })
  }
})

from fastapi import APIRouter
import os

router = APIRouter(prefix="/files")

BASE_PATH = "./KNOWLEDGE"

@router.get("/list")
def list_files(user_name: str, path: str = ""):

    target = os.path.join(BASE_PATH, user_name, path)

    if not os.path.exists(target):
        return {"ok": True, "data": []}

    items = []

    for name in os.listdir(target):
        full = os.path.join(target, name)

        items.append({
            "name": name,
            "isDir": os.path.isdir(full),
            "path": f"{path}/{name}" if path else name
        })

    return {"ok": True, "data": items}


@router.get("/open")
def open_file(user_name: str, path: str):

    full = os.path.join(BASE_PATH, user_name, path)

    if not os.path.exists(full):
        return {"ok": False, "error": "not found"}

    with open(full, "r", encoding="utf-8") as f:
        content = f.read()

    return {"ok": True, "content": content}

from fastapi import APIRouter
import requests
import os

router = APIRouter(prefix="/auth")

SUPABASE_URL = os.getenv("SUPABASE_URL")
SUPABASE_KEY = os.getenv("SUPABASE_KEY")

@router.post("/login")
def login(payload: dict):

    email = payload["email"]
    password = payload["password"]

    r = requests.post(
        f"{SUPABASE_URL}/auth/v1/token?grant_type=password",
        json={"email": email, "password": password},
        headers={"apikey": SUPABASE_KEY}
    )

    if r.status_code != 200:
        return {"ok": False, "error": "login failed"}

    user = r.json().get("user")

    return {
        "ok": True,
        "user": {
            "id": user["id"],
            "email": user["email"],
            "user_name": email
        }
    }
